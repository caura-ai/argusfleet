\begin{column}{0.48\textwidth}

\textbf{Thesis.} AI memory is shifting from a context-window problem to
a distributed-systems problem. We propose four governed-memory primitives,
instantiate them in \textbf{MemClaw} (\texttt{memclaw.net}) and evaluate via
\textbf{ArgusFleet} --- 4 experiments $\to$ 5 metric axes against a live
service.

\vspace{0.3em}

\textbf{Architectural findings}
\begin{enumerate}\setlength\itemsep{0.1em}\setlength\parskip{0pt}\setlength\leftmargin{1em}
  \item \textcolor{warn}{\textbf{Scope enforcement was bimodal} (now fixed).}
        Sub-tenant scope was unenforced on \texttt{GET}-by-id (handler
        discarded a resolved agent identity); disclosed \&
        \textcolor{good}{remediated}.
  \item \textcolor{warn}{\textbf{Dedup starves contradiction detection.}}
        Synchronous embedding \texttt{409}s pre-empt the async detector.
        Detector is sound (90/90 when both land); the \emph{ordering} defeats it.
  \item \textcolor{warn}{\textbf{Cross-agent propagation is privilege-gated.}}
        \texttt{redistribute} requires \texttt{trust\_level $\ge$ 3},
        conflating fleet-internal propagation with admin migration.
  \item \textcolor{good}{\textbf{Consistency cost is at write time.}}
        Strong-mode write$\to$visible is $\sim$0.8 s (synchronous); the
        cost is write latency, not a read tail.
\end{enumerate}

\end{column}

\textbf{Take-aways}
\begin{itemize}\setlength\itemsep{0.05em}\setlength\parskip{0pt}\setlength\leftmargin{1em}
  \item GET-by-id gap (164/164 at meas.) ---
        \textcolor{good}{disclosed \& remediated}; agent-scoped GET now
        enforces fleet+trust (0/36).
  \item Provenance: \textcolor{good}{sub-second}, \textcolor{good}{1.00 completeness + accuracy}.
  \item Propagation: \textcolor{good}{0 leaks at $n=80$};
        \textcolor{good}{$\sim$0.8 s window} (synchronous enrichment).
  \item Contradiction: \textcolor{good}{sound when admitted (90/90)};
        \textcolor{warn}{dedup gate starves it} (0.49 overall).
\end{itemize}
