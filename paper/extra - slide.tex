\begin{column}{0.48\textwidth}

\textbf{Thesis.} AI memory is shifting from a context-window problem to
a distributed-systems problem: governance, synchronization, temporal
correctness, provenance, scoped access. We propose four primitives,
instantiate them in \textbf{MemClaw} (\texttt{memclaw.net}) and evaluate
via \textbf{ArgusFleet} --- 4 experiments $\to$ 5 metric axes against a
live service. Failure modes: leakage, stale propagation, contradiction
persistence, provenance collapse.

\vspace{0.3em}

\textbf{Three architectural findings}
\begin{enumerate}\setlength\itemsep{0.1em}\setlength\parskip{0pt}\setlength\leftmargin{1em}
  \item \textcolor{warn}{\textbf{Scope enforcement is bimodal.}}
        Tenant uniformly; sub-tenant only on search, not on
        \texttt{GET}-by-id --- tenant-scoped keys carry no agent identity.
        A property of the auth model, not a server defect.
  \item \textcolor{warn}{\textbf{Dedup precedes contradiction detection.}}
        Embedding \texttt{409}s block detection even on a clean tenant.
        Pipeline ordering should invert when an RDF triple is present.
  \item \textcolor{warn}{\textbf{Cross-agent propagation is privilege-gated.}}
        \texttt{redistribute} requires \texttt{trust\_level $\ge$ 3},
        conflating fleet-internal propagation with admin migration.
\end{enumerate}

\end{column}

\textbf{Take-aways}
\begin{itemize}\setlength\itemsep{0.05em}\setlength\parskip{0pt}\setlength\leftmargin{1em}
  \item Within-tenant scope: \textcolor{warn}{asymmetric on GET vs search}.
  \item Provenance: \textcolor{good}{sub-second}, \textcolor{good}{1.00 accuracy}.
  \item Propagation: \textcolor{good}{0 leaks at $n=40$},
        \textcolor{warn}{$\sim$80 s window}.
  \item Contradiction null is \textcolor{warn}{architectural} --- intra-run dedup.
\end{itemize}
