\begin{column}{0.48\textwidth}

\textbf{Thesis.} AI memory is shifting from a context-window problem to a
distributed-systems problem: governance, synchronization, temporal
correctness, provenance, scoped access.

\medskip

\textbf{Four failure modes we identify and measure}
\begin{itemize}\setlength\itemsep{0.1em}
  \item Unauthorized leakage
  \item Stale propagation
  \item Contradiction persistence
  \item Provenance collapse
\end{itemize}

\medskip

\textbf{What we built}
\begin{itemize}\setlength\itemsep{0.1em}
  \item \textbf{MemClaw} --- managed governed-memory service in production
        at \texttt{memclaw.net}.
  \item \textbf{AgentFleetMem} --- open, async, reproducible evaluation
        harness. 4 experiments $\to$ 5 metric axes. 33 tests.
\end{itemize}

\medskip

\textbf{Three architectural findings}
\begin{enumerate}\setlength\itemsep{0.1em}
  \item \textcolor{warn}{\textbf{Scope enforcement is bimodal.}}
        Tenant-level uniformly; sub-tenant only on the search path,
        not on direct \texttt{GET}-by-id. A function of the
        authentication model, not a server defect.
  \item \textcolor{warn}{\textbf{Dedup precedes contradiction detection.}}
        Embedding-similarity \texttt{409}s block contradictions even
        on a clean tenant where no historical near-duplicates exist.
  \item \textcolor{warn}{\textbf{Cross-agent propagation is privilege-gated.}}
        \texttt{redistribute} requires \texttt{trust\_level $\ge$ 3} ---
        conflates fleet-internal propagation with admin migration.
\end{enumerate}

\end{column}

\medskip

\textbf{What the data supports}
\begin{itemize}\setlength\itemsep{0.1em}
  \item Tenant-level scope \textcolor{good}{enforced uniformly};
        within-tenant scope is \textcolor{warn}{enforced asymmetrically}
        across GET and search paths.
  \item Provenance is a \textcolor{good}{viable architectural primitive}
        at sub-second per-hop latency.
  \item Cross-agent propagation: \textcolor{good}{zero observed cross-fleet
        leakage} across 40 probes, with a
        \textcolor{warn}{$\sim$80 s eventual-consistency window}.
  \item Production memory pipelines need architectural attention to
        ordering, the dedup-vs-contradiction gate, and propagation privilege.
\end{itemize}
